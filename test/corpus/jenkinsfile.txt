    stage('Build') {
        git url: 'https://github.com/org/repo.git', branch: 'main'
        sh './gradlew assemble'
    }
    stage('Test') {
        try {
            sh './gradlew test'
        } catch (Exception err) {
            echo "Caught error"
            currentBuild.result = 'FAILURE'
        } finally {
            junit 'build/test-results/**/*.xml'
        }
    }
}
